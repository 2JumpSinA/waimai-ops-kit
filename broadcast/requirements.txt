# ── 必需（不装 = 取不到余额）────────────────────────────────────────────
requests>=2.31,<3
    # 干什么：所有 HTTP 请求。淘宝闪购的 MTOP 签名调用、饿了么 AGW 余额接口、
    #         以及从浏览器 profile 读登录态都走它。
    # 出处：  platforms/ele.py:44、core/profile_cookie.py:35（模块级 import，无 try 兜底）

websocket-client>=1.6,<2
    # 干什么：CDP（Chrome DevTools Protocol）WebSocket 客户端 —— 没有它就驱动不了
    #         浏览器，美团余额只能靠页面内签名取数，整条链路不可用。
    # 注意：  包名是 websocket-client，代码里 import 的名字是 websocket（这是该包的
    #         历史命名，不是笔误）。
    # 出处：  core/browser.py:34（模块级 import）、core/profile_cookie.py:71

# ── 可选（不装 = 自动降级，不崩）────────────────────────────────────────
keyring>=24
    # 干什么：把 Cookie 凭据存进操作系统凭据管理器（Windows Credential Manager）。
    # 不装会怎样： core/store.py:46-51 是 try/except ImportError —— keyring 缺失时
    #             _HAS_KEYRING=False，凭据改存 creds/<cred_key>.txt 明文文件，
    #             并在日志里打一条 WARN。功能可用，安全性下降。
    # 什么时候必装： 用 plan 任务/无人值守跑，且希望凭据不落明文盘时。
    # 出处：  core/store.py:47
